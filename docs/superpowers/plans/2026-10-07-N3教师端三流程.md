```java
package com.fz.scitech.hx;

import com.fz.scitech.hx.entity.*;
import com.fz.scitech.platform.template.RmbCapitalUtil;
import com.fz.scitech.system.entity.SysDict;
import com.fz.scitech.system.mapper.SysDictMapper;
import com.baomidou.mybatisplus.core.conditions.query.LambdaQueryWrapper;
import org.springframework.stereotype.Component;

import java.math.BigDecimal;
import java.math.RoundingMode;
import java.time.LocalDate;
import java.time.format.DateTimeFormatter;
import java.util.*;
import java.util.stream.Collectors;

/**
 * rp_* 实体 → poi-tl 渲染 model。口径与 SampleData 完全一致(勾选组 √/□、大写金额、
 * 日期 yyyy年MM月、万元两位小数、成员 m/学生 s 连续非空平铺契约,超预置由 expandDynamicRows 扩)。
 */
@Component
public class RenderModelBuilder {

    private static final DateTimeFormatter YM = DateTimeFormatter.ofPattern("yyyy年MM月");
    /** 预算科目→登记表占位符(占位符清单.md) */
    private static final Map<String, String> BUDGET_KEY = Map.of(
            "业务费", "budgetBusiness", "劳务费", "budgetLabor", "绩效", "budgetPerformance",
            "业务接待费", "budgetReception", "协作费", "budgetCollab", "其它", "budgetOther");

    private final SysDictMapper dictMapper;

    public RenderModelBuilder(SysDictMapper dictMapper) {
        this.dictMapper = dictMapper;
    }

    private List<String> dictLabels(String dictType) {
        return dictMapper.selectList(new LambdaQueryWrapper<SysDict>()
                        .eq(SysDict::getDictType, dictType).eq(SysDict::getStatus, 1)
                        .orderByAsc(SysDict::getSortOrder))
                .stream().map(SysDict::getDictLabel).collect(Collectors.toList());
    }

    /** 勾选组:selected 命中 √,其余 □(顺序=字典顺序,与模板一致) */
    static String checks(List<String> options, String... selected) {
        Set<String> hit = new HashSet<>(Arrays.asList(selected));
        StringBuilder sb = new StringBuilder();
        for (String o : options) {
            if (sb.length() > 0) {
                sb.append(' ');
            }
            sb.append(hit.contains(o) ? "√" : "□").append(o);
        }
        return sb.toString();
    }

    static String startEndDate(LocalDate s, LocalDate e) {
        return (s == null ? "" : YM.format(s)) + "至" + (e == null ? "" : YM.format(e));
    }

    static String wan(BigDecimal yuan) {
        return yuan == null ? "" : yuan.movePointLeft(4).setScale(2, RoundingMode.HALF_UP).toPlainString();
    }

    static String money(BigDecimal yuan) {
        return yuan == null ? "" : yuan.setScale(2, RoundingMode.HALF_UP).toPlainString();
    }

    public Map<String, Object> buildRegister(RpProject p, RpHxProject hx, List<RpMember> members, List<RpBudget> budgets) {
        Map<String, Object> m = base(p);
        m.put("projectTypeChecks", checks(dictLabels("hx_project_type"), hx.getProjectType()));
        m.put("startEndDate", startEndDate(hx.getStartDate(), hx.getEndDate()));
        m.put("clientName", s(hx.getClientName()));
        m.put("clientNatureChecks", checks(dictLabels("client_nature"), hx.getClientNature()));
        m.put("contractAmount", money(hx.getContractAmount()));
        m.put("contractAmountCapital", hx.getContractAmount() == null ? "" : RmbCapitalUtil.toCapital(hx.getContractAmount()));
        // leaderName/deptName 在 base();电话邮箱:
        m.put("leaderEmail", s(hx.getLeaderEmail()));
        m.put("projectSummary", s(hx.getProjectSummary()));
        m.put("progressPlan", s(hx.getProgressPlan()));
        Map<String, String> budgetMap = budgets.stream()
                .collect(Collectors.toMap(RpBudget::getSubject, b -> money(b.getAmount()), (a, b) -> a));
        BUDGET_KEY.forEach((subject, key) -> m.put(key, budgetMap.getOrDefault(subject, "")));
        m.put("useSchoolAssetChecks", checks(List.of("是", "否"),
                hx.getUseSchoolAsset() != null && hx.getUseSchoolAsset() == 1 ? "是" : "否"));
        boolean charge = hx.getChargeMgmtFee() != null && hx.getChargeMgmtFee() == 1;
        m.put("chargeMgmtFeeChecks", checks(List.of("是", "否"), charge ? "是" : "否"));
        m.put("mgmtFeeAmount", charge ? money(hx.getMgmtFeeAmount()) : "");
        m.put("hxRegisterDate", ""); // 科技处确认时才写(N4 终版渲染)
        putRows(m, "m", members.stream().filter(x -> !"学生".equals(x.getMemberType())).collect(Collectors.toList()),
                "name", "unit", "title", "major", "duty");
        return m;
    }

    public Map<String, Object> buildChange(RpProject p, RpHxProject hx, RpHxFlow flow, String leaderPhone) {
        Map<String, Object> m = base(p);
        m.put("projectTypeChecks", checks(dictLabels("hx_project_type"), hx.getProjectType()));
        m.put("leaderPhone", s(leaderPhone));
        String[] types = flow.getChangeType() == null ? new String[0] : flow.getChangeType().split(",");
        m.put("changeTypeChecks", checks(dictLabels("change_type"), types));
        m.put("changeContent", s(flow.getChangeContent()));
        m.put("changeReason", s(flow.getChangeReason()));
        return m;
    }

    public Map<String, Object> buildAcceptance(RpProject p, RpHxProject hx, RpHxFlow flow,
                                               List<RpMember> members, BigDecimal receivedTotal) {
        Map<String, Object> m = base(p);
        m.put("projectTypeChecks", checks(dictLabels("hx_project_type"), hx.getProjectType()));
        m.put("startEndDate", startEndDate(hx.getStartDate(), hx.getEndDate()));
        m.put("clientName", s(hx.getClientName()));
        m.put("clientNatureChecks", checks(dictLabels("client_nature"), hx.getClientNature()));
        m.put("receivedAmountWan", wan(receivedTotal));
        m.put("usedAmountWan", wan(flow.getUsedAmount()));
        List<RpMember> teachers = members.stream().filter(x -> !"学生".equals(x.getMemberType())).collect(Collectors.toList());
        putRows(m, "m", teachers, "name", "unit", "title", "major", "dutyRatio");
        for (int i = 0; i < teachers.size(); i++) {
            RpMember t = teachers.get(i);
            String ratio = t.getRatio() == null ? "" : t.getRatio().setScale(2, RoundingMode.HALF_UP).toPlainString();
            String amtWan = t.getAmount() == null ? "" : wan(t.getAmount());
            m.put("m" + (i + 1) + "dutyRatio",
                    (ratio.isEmpty() && amtWan.isEmpty()) ? "" : ratio + "%(" + amtWan + "万元)");
        }
        putRows(m, "s", members.stream().filter(x -> "学生".equals(x.getMemberType())).collect(Collectors.toList()),
                "name", "studentNo", "major", "mentor", "duty");
        m.put("finishDate", ""); // 验收确认日,N4 终版渲染
        m.put("outcomeFormChecks", checks(dictLabels("hx_outcome_form"),
                flow.getOutcomeForm() == null ? new String[0] : flow.getOutcomeForm().split(",")));
        m.put("acceptSummary", s(flow.getAcceptSummary()));
        m.put("verifiedAmountWan", ""); // 核定 N4 录入
        return m;
    }

    public Map<String, Object> buildReport(RpProject p, RpHxFlow flow) {
        Map<String, Object> m = base(p);
        m.put("onTimeChecks", checks(List.of("是", "否"),
                flow.getIsOnTime() != null && flow.getIsOnTime() == 1 ? "是" : "否"));
        m.put("projectTypeChecks", ""); // 报告表无此占位符则无需;见占位符清单 REPORT 集
        m.put("outcomeFormChecks", checks(dictLabels("hx_outcome_form"),
                flow.getOutcomeForm() == null ? new String[0] : flow.getOutcomeForm().split(",")));
        m.put("reportParagraphs", flow.getReportContent() == null || flow.getReportContent().isBlank()
                ? List.of() : Arrays.stream(flow.getReportContent().split("\\n\\n"))
                        .map(String::trim).filter(x -> !x.isEmpty()).collect(Collectors.toList()));
        return m;
    }

    private Map<String, Object> base(RpProject p) {
        Map<String, Object> m = new LinkedHashMap<>();
        m.put("approvalNo", s(p.getApprovalNo()));
        m.put("projectName", s(p.getName()));
        m.put("leaderName", s(p.getLeaderName()));
        m.put("deptName", s(p.getDeptName())); // RpProject 需补 @TableField(exist=false) deptName 或 N4 前 join;见 Step5 注
        return m;
    }

    /** 固定行平铺(SampleData.putFixedRows 同款):连续非空行 i=1..size,尾部到 size 即止(空行由模板预置渲染空) */
    private static void putRows(Map<String, Object> m, String prefix, List<RpMember> people, String... fields) {
        // 仅平铺实际条数(契约:调用方保证顺序=sort_no;expandDynamicRows 按连续非空计数)
        for (int i = 0; i < people.size(); i++) {
            RpMember p = people.get(i);
            for (String f : fields) {
                m.put(prefix + (i + 1) + f, val(p, f));
            }
        }
    }

    private static String val(RpMember p, String field) {
        return switch (field) {
            case "name" -> s(p.getName());
            case "unit" -> s(p.getUnit());
            case "title" -> s(p.getTitle());
            case "major" -> s(p.getMajor());
            case "duty" -> s(p.getDuty());
            case "studentNo" -> s(p.getStudentNo());
            case "mentor" -> s(p.getMentor());
            default -> "";
        };
    }

    private static String s(String v) {
        return v == null ? "" : v;
    }
}
```

```java
/**

 * 教师越权防线=每条查询/更新强制 apply_user_id 条件(spec §6.4)。

@Service
public class HxService {

    /** 幂等:flowId 空新建(kind=2 草稿行+flow register 草稿行),非空整体覆盖(仅 status=0) */
    @Transactional
    public Long saveRegisterDraft(Long uid, String username, RegisterSaveDTO dto) {
        validateRegister(dto);
        Long flowId;
        RpProject project;
        if (dto.getFlowId() == null) {
            project = new RpProject();
            project.setKind(2);
            project.setAuditStatus(0);
            project.setApplyUserId(uid);
            projectMapper.insert(project);
            RpHxFlow flow = new RpHxFlow();
            flow.setProjectId(project.getId());
            flow.setFlowType("register");
            flow.setStatus(HxFlowStatus.DRAFT);
            flow.setCreateBy(username);
            flowMapper.insert(flow);
            flowId = flow.getId();
        } else {
            RpHxFlow flow = ownedFlow(uid, dto.getFlowId());
            if (flow.getStatus() != HxFlowStatus.DRAFT) {
                throw new BizException("仅草稿可编辑");
            }
            flowId = flow.getId();
            project = projectMapper.selectById(flow.getProjectId());
        }
        project.setName(dto.getName());
        project.setLeaderName(dto.getLeaderName());
        project.setDeptId(dto.getDeptId());
        projectMapper.updateById(project);
        RpHxProject hx = hxMapper.selectById(project.getId());
        if (hx == null) {
            hx = new RpHxProject();
            hx.setProjectId(project.getId());
        }
        copyHx(dto.getHx(), hx);
        if (hxMapper.selectById(project.getId()) == null) {
            hxMapper.insert(hx);
        } else {
            hxMapper.updateById(hx);
        }
        replaceMembers(project.getId(), dto.getMembers());
        replaceBudgets(project.getId(), dto.getBudgets());
        return flowId;

    /** 必填校验:名称/负责人/院部/类型/委托单位/性质/经费/起止;成员姓名非空 */
    private void validateRegister(RegisterSaveDTO dto) {
        if (isBlank(dto.getName()) || isBlank(dto.getLeaderName()) || dto.getDeptId() == null) {
            throw new BizException("项目名称/负责人/院部必填");
        }
        RegisterSaveDTO.Hx h = dto.getHx();
        if (h == null || isBlank(h.getProjectType()) || isBlank(h.getClientName())
                || isBlank(h.getClientNature()) || h.getContractAmount() == null
                || isBlank(h.getStartDate()) || isBlank(h.getEndDate())) {
            throw new BizException("项目类型/委托单位/性质/合同经费/起止时间必填");
        }

    // ---------- 提交(生成待签双出) ----------
    @Transactional

        RpHxFlow flow = ownedFlow(uid, flowId);
        if (flow.getStatus() != HxFlowStatus.DRAFT) {
            throw new BizException("仅草稿可提交");
        }
        RegisterSaveDTO dto = loadRegisterDto(flow.getProjectId()); // 反查实体拼 DTO 复用校验
        validateRegister(dto);
        transit(flowId, HxFlowStatus.DRAFT, HxFlowStatus.SUBMITTED);
        flowMapper.update(null, new LambdaUpdateWrapper<RpHxFlow>()
                .eq(RpHxFlow::getId, flowId).set(RpHxFlow::getSubmitTime, LocalDateTime.now()));
        projectMapper.update(null, new LambdaUpdateWrapper<RpProject>()
                .eq(RpProject::getId, flow.getProjectId())
                .set(RpProject::getAuditStatus, 1)
                .set(RpProject::getDeclareDate, LocalDate.now()));
        renderDeliverables(flow, uid, username);   // 双出 docx+pdf(见下)
        saveRecord(flow, HxFlowStatus.ACT_SUBMIT, username, uid, null);
    }

    /** 渲染待签交付物:render 双出落盘+附件登记;旧版 hx_docx/hx_pdf 标记作废 */
    private void renderDeliverables(RpHxFlow flow, Long uid, String username) {
        invalidateOld(flow.getId(), flow.getProjectId(), "hx_docx", "hx_pdf");
        RpProject p = projectMapper.selectById(flow.getProjectId());
        fillDeptName(p);
        RpHxProject hx = hxMapper.selectById(flow.getProjectId());
        Map<String, Object> model;
        String tplType;
        if ("register".equals(flow.getFlowType())) {
            model = modelBuilder.buildRegister(p, hx,
                    memberMapper.selectList(new LambdaQueryWrapper<RpMember>()
                            .eq(RpMember::getProjectId, flow.getProjectId()).orderByAsc(RpMember::getSortNo)),
                    budgetMapper.selectList(new LambdaQueryWrapper<RpBudget>()
                            .eq(RpBudget::getProjectId, flow.getProjectId())));
            tplType = "hx_register";
        } else {
            throw new BizException("未知流程类型:" + flow.getFlowType()); // change/acceptance 由 Task21 扩展
        }
        // 注意 operatorId 必传 uid(附件 uploader_id 落库;Task17 RenderContext 六参,勿传 null)
        pdfRenderService.render(new com.fz.scitech.platform.template.dto.RenderContext(
                flow.getId(), flow.getProjectId(), flow.getFlowType(), tplType,
                uid, username, false), model);
    }

    /**
     * 同一流程单旧待签件作废(逻辑删附件行;文件保留不入列表=设计§3.2)。
     * RpAttachment 无 flowId 列,按 file_path LIKE '%/{flowId}/%' 定位(渲染件与扫描件路径均含 /{fid}/ 段,
     * 斜杠边界保证 flowId=8 不误配 /18/);biz_type IN (...) 收窄。
     */
    private void invalidateOld(Long flowId, Long projectId, String... bizTypes) {
        List<RpAttachment> olds = attachmentMapper.selectList(new LambdaQueryWrapper<RpAttachment>()
                .eq(RpAttachment::getProjectId, projectId)
                .in(RpAttachment::getBizType, Arrays.asList(bizTypes))
                .likeRight(RpAttachment::getFilePath, "hx/" + projectId + "/"));
        for (RpAttachment a : olds) {
            if (a.getFilePath().contains("/" + flowId + "/")) {
                attachmentMapper.deleteById(a.getId()); // 逻辑删(deleted=1)
            }
        }
    }

    // ---------- 撤回(三流程共用) ----------
    @Transactional
    public void withdraw(Long uid, Long flowId) {
        RpHxFlow flow = ownedFlow(uid, flowId);
        if (flow.getStatus() == HxFlowStatus.SUBMITTED) {
            transit(flowId, 1, 0);
        } else if (flow.getStatus() == HxFlowStatus.SCANNED) {
            transit(flowId, 2, 1);
        } else {
            throw new BizException("当前状态不可撤回");
        }
    }

    /** doc_type 按单据四选一(与 flow_type 对应校验);每类仅最新一份生效,重传旧件作废 */

        // 旧 hx_scan 同 docType 作废(deleted=1)

        flowMapper.update(... set scanTime=now ...);

    /** 教师本人流程单(越权防线;BizException 单参,消息文本区分 404/403 语义) */

        }
```
